// Switch to the correct database
use sample_mflix

// ---------------------------------------------
// Example 1: Match by multiple genres
// Task: Find movies that are either Drama or Comedy
// ---------------------------------------------

db.movies.find(
  { genres: { $in: ["Drama", "Comedy"] } },
  { title: 1, genres: 1, _id: 0 }
).limit(5)


// ---------------------------------------------
// Example 2: Negation with $nin
// Task: Exclude movies that are Action or Horror
// ---------------------------------------------

db.movies.find(
  { genres: { $nin: ["Action", "Horror"] } },
  { title: 1, genres: 1, _id: 0 }
).limit(5)


// ---------------------------------------------
// Example 3: Numeric $in
// Task: Find movies released in 1999, 2005, or 2010
// ---------------------------------------------

db.movies.find(
  { year: { $in: [1999, 2005, 2010] } },
  { title: 1, year: 1, _id: 0 }
).limit(5)


// ---------------------------------------------
// Example 4: Regex with $in
// Task: Match movies where cast includes names starting with "Harry" or "Will"
// ---------------------------------------------

db.movies.find(
  { cast: { $in: [/^Harry/, /^Will/] } },
  { title: 1, cast: 1, _id: 0 }
).limit(5)


// ---------------------------------------------
// Example 5: Logical AND with $in
// Task: Find English-language movies that are also Comedy or Drama
// ---------------------------------------------

db.movies.find(
  { languages: "English", genres: { $in: ["Comedy", "Drama"] } },
  { title: 1, languages: 1, genres: 1, _id: 0 }
).limit(5)
