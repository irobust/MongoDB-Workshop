// Switch to the correct database
use sample_mflix

// ---------------------------------------------
// Example 0: How genres look before unwind
// Task: Show movies with their genres array intact
// ---------------------------------------------
db.movies.find(
  {},
  { title: 1, genres: 1, _id: 0 }
).limit(5)


// ---------------------------------------------
// Example 1: Basic $unwind on genres
// Task: Create one row per genre for each movie
// ---------------------------------------------
db.movies.aggregate([
  { $unwind: "$genres" },
  { $project: { _id: 0, title: 1, genre: "$genres" } },
  { $limit: 10 }
])

// ---------------------------------------------
// Example 2: Count movies per genre
// Task: How many movies belong to each genre
// ---------------------------------------------
db.movies.aggregate([
  { $unwind: "$genres" },
  { $group: { _id: "$genres", movieCount: { $sum: 1 } } },
  { $sort: { movieCount: -1, _id: 1 } },
  { $limit: 10 }
])

// ---------------------------------------------
// Example 3: Average rating per genre
// Task: Unwind genres, then compute avg imdb.rating by genre
// ---------------------------------------------
db.movies.aggregate([
  { $unwind: "$genres" },
  { $group: {
      _id: "$genres",
      avgRating: { $avg: "$imdb.rating" },
      titles: { $sum: 1 }
  } },
  { $project: { _id: 0, genre: "$_id", titles: 1, avgRating: { $round: ["$avgRating", 2] } } },
  { $sort: { avgRating: -1, titles: -1 } },
  { $limit: 10 }
])

// ---------------------------------------------
// Example 4: Year-by-year genre trend
// Task: Count movies per genre per release year
// Notes: Ensure released is a date to avoid errors with $year
// ---------------------------------------------
db.movies.aggregate([
  { $match: { released: { $type: "date" } } },
  { $unwind: "$genres" },
  { $project: { _id: 0, genre: "$genres", year: { $year: "$released" } } },
  { $group: { _id: { genre: "$genre", year: "$year" }, movieCount: { $sum: 1 } } },
  { $sort: { "_id.genre": 1, "_id.year": 1 } },
  { $limit: 20 }
])
