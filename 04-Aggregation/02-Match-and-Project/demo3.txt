// Switch to the correct database
use sample_mflix

// ---------------------------------------------
// Example 1: Rename a field
// Task: Show movies after 2010 with year renamed to releaseYear
// ---------------------------------------------

db.movies.aggregate([
  { $match: { year: { $gt: 2010 } } },
  { $project: { title: 1, releaseYear: "$year", _id: 0 } },
  { $limit: 5 }
])


// ---------------------------------------------
// Example 2: Add a computed decade field
// Task: Show movies after 1998 with a decade field based on year
// ---------------------------------------------

db.movies.aggregate([
  { $match: { year: { $gte: 1998 } } },
  { $project: { 
      title: 1,
      year: 1,
      decade: { $multiply: [ { $floor: { $divide: ["$year", 10] } }, 10 ] },
      _id: 0
    }
  },
  { $limit: 20 }
])


// ---------------------------------------------
// Example 3: Combine rename + computed fields
// Task: Show movies after 1998 with uppercase title, releaseYear, and decade
// ---------------------------------------------

db.movies.aggregate([
  { $match: { year: { $gt: 1998 } } },
  { $project: { 
      movieTitle: { $toUpper: "$title" },
      releaseYear: "$year",
      decade: { $multiply: [ { $floor: { $divide: ["$year", 10] } }, 10 ] },
      _id: 0
    }
  },
  { $limit: 10 }
])


// ---------------------------------------------
// Example 4: Conditional computed field
// Task: Show movies after 1990 with a flag isModern ("Yes" if year > 2000, else "No")
// ---------------------------------------------

db.movies.aggregate([
  { $match: { year: { $gte: 1990 } } },
  { $project: { 
      title: 1,
      year: 1,
      isModern: { $cond: [ { $gt: ["$year", 2000] }, "Yes", "No" ] },
      _id: 0
    }
  },
  { $limit: 10 }
])
