// Switch to the correct database
use sample_mflix

// ---------------------------------------------
// Example 1: Single stage filter
// Task: Select only movies released after 2010
// ---------------------------------------------

db.movies.aggregate([
  { $match: { year: { $gt: 2010 } } }
])


// ---------------------------------------------
// Example 2: Filter + reshape
// Task: After filtering, keep only title and year
// ---------------------------------------------

db.movies.aggregate([
  { $match: { year: { $gt: 2010 } } },
  { $project: { title: 1, year: 1, _id: 0 } }
])


// ---------------------------------------------
// Example 3: Add a computed field
// Task: Show title, year, and also create a label field
// ---------------------------------------------

db.movies.aggregate([
  { $match: { year: { $gt: 2010 } } },
  { $project: { 
      title: 1,
      year: 1,
      label: { $concat: ["Title: ", "$title", " (", { $toString: "$year" }, ")"] },
      _id: 0
    } 
  }
])


// ---------------------------------------------
// Example 4: Multi-stage with computed filter
// Task: Filter by year, compute titleLength, then filter again
// ---------------------------------------------

db.movies.aggregate([
  { $match: { year: { $gt: 2010 } } },
  { $project: { 
      title: 1,
      year: 1,
      titleLength: { $strLenCP: "$title" },
      _id: 0 
    } 
  },
  { $match: { titleLength: { $gt: 30 } } }
])
