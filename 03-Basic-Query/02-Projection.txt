// Step 2: 1994 movies, but hide _id

  { year: 1994 },
  { _id: 0 }

// Step 3: 1994 movies, hide _id & include title & director
db.movies.find(
  { year: 1994 },
  { _id: 0, title: 1, director: 1 }
);

// Step 4: 1994 hits, hide _id & include title, year & runtime
db.movies.find(
  { year: 1994 },
  { _id: 0, title: 1, year: 1, runtime: 1 }
);

// Step 5: Aggregation – Top 5 Directors in 1994
db.movies.aggregate([
  { $match: { year: 1994 } },
  { $unwind: "$directors" },
  { $group: {
      _id: "$directors",
      count: { $sum: 1 }
  }},
  { $sort: { count: -1 } },
  { $limit: 5 },
  { $project: {
      _id:      0,
      director: "$_id",
      count:    1
  }}
]);

// # Demo: Including or Excluding Fields in Result Sets (mongosh)

// Step 1: Default—every field, every nested bit
db.movies.find(
  {}
);

// Step 2: Inclusion—only title, year & genres
db.movies.find(
  {},
  { title: 1, year: 1, genres: 1 }
);

// Step 3: Exclusion—hide plot and awards
db.movies.find(
  {},
  { plot: 0, awards: 0 }
);

// Step 4: Array slice—first 3 cast members, drop _id
db.movies.find(
  {},
  { cast: { $slice: 3 }, _id: 0 }
);

// Step 5: Include title and only ratings with score > 8.5
db.movies.find(
  {},
  {
    title:   1,
    ratings: { $elemMatch: { score: { $gt: 8.5 } } },
    _id:     0
  }
);

// Step 6: Show full cast + last 2 cast members
db.movies.aggregate([
  {
    $project: {
      _id:      0,
      title:    1,
      cast:     1,
      lastCast: { $slice: ["$cast", -2] }
    }
  }
]);
