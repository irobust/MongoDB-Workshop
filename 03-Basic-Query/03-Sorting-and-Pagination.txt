// # Demo: Sorting by Numeric, Date, and String Fields - Shell

use sample_mflix

// Step 1: Sort by runtime (numeric field)
db.movies.find(
  { year: { $gte: 2007 } },
  { title: 1, runtime: 1 }
).sort({ runtime: 1 })

// Step 2: Sort by runtime (numeric field) descending
db.movies.find(
  { year: { $gte: 2007 } },
  { title: 1, runtime: 1 }
).sort({ runtime: -1 })

// Step 3: Sort only when runtime exists (field is present and valid)
db.movies.find(
  { year: { $gte: 2007 }, runtime: { $exists: true } },
  { title: 1, runtime: 1 }
).sort({ runtime: 1 })

// Step 4: Sort titles using numeric-aware collation
// Helps correctly order titles like "Movie 2" before "Movie 10"
db.movies.find(
  { title: { $exists: true } },
  { title: 1 }
).sort({ title: 1 }).collation({ locale: "en", numericOrdering: true })

// Step 5: Sort by nested subdocument field (awards.wins)
db.movies.find(
  { year: { $gte: 2007 }, "awards.wins": { $exists: true } },
  { title: 1, "awards.wins": 1 }
).sort({ "awards.wins": -1 })

// Step 6: Sort movies by second language listed in the 'languages' array
// Uses aggregation pipeline to extract second language and sort
db.movies.aggregate([
  {
    $match: {
      year: { $gte: 2007 },
      "languages.1": { $exists: true }
    }
  },
  {
    $addFields: {
      second_lang: { $arrayElemAt: ["$languages", 1] }
    }
  },
  {
    $sort: { second_lang: -1 }
  },
  {
    $project: {
      title: 1,
      languages: 1,
      second_lang: 1
    }
  }
])
